// 滚动动画（增加轻微缩放）
function setupScrollAnimations() {
  const sections = document.querySelectorAll('.fade-in-section');
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const items = entry.target.querySelectorAll(':scope > .section-title, :scope > .section-desc, :scope > .product-grid, :scope > .split');
        items.forEach((el, index) => {
          el.style.transition = 'opacity 0.8s cubic-bezier(0.16,1,0.3,1), transform 0.8s cubic-bezier(0.16,1,0.3,1)';
          el.style.transitionDelay = `${index * 0.1}s`;
          el.style.opacity = '1';
          el.style.transform = 'translateY(0) scale(1)';
        });
        entry.target.classList.add('is-visible');
      }
    });
  }, { threshold: 0.1, rootMargin: "0px 0px -30px 0px" });

  sections.forEach(section => {
    const items = section.querySelectorAll(':scope > .section-title, :scope > .section-desc, :scope > .product-grid, :scope > .split');
    items.forEach(el => {
      el.style.opacity = '0';
      el.style.transform = 'translateY(24px) scale(0.98)';
    });
    observer.observe(section);
  });
}

function setupLogoScale() {
  const logo = document.querySelector('.logo');
  window.addEventListener('scroll', () => {
    const scale = Math.max(0.85, 1 - window.scrollY / 1500);
    logo.style.transform = `scale(${scale})`;
    logo.style.fontWeight = window.scrollY > 100 ? '500' : '600';
  });
}

function setupNavScroll() {
  const nav = document.getElementById('navbar');
  window.addEventListener('scroll', () => {
    if (window.scrollY > 20) nav.classList.add('scrolled');
    else nav.classList.remove('scrolled');
  });
}

function setupSmoothNav() {
  const dropdown = document.getElementById('productsDropdown');
  const links = dropdown.querySelectorAll('.nav-link');
  links.forEach(link => {
    link.addEventListener('click', function(e) {
      e.preventDefault();
      const targetId = this.getAttribute('href').substring(1);
      const target = document.getElementById(targetId);
      if (target) {
        dropdown.classList.add('force-hide');
        setTimeout(() => {
          target.scrollIntoView({ behavior: 'smooth' });
          dropdown.classList.remove('force-hide');
        }, 150);
      }
    });
  });
}

function renderAllProducts() {
  Object.keys(brandIntro).forEach(brand => {
    const section = document.querySelector(`#${getBrandId(brand)} .product-grid`);
    const desc = document.querySelector(`#${getBrandId(brand)} .section-desc`);
    if(desc) desc.textContent = brandIntro[brand];
    const brandProducts = products.filter(p => p.brand === brand);
    if(section) {
      section.innerHTML = brandProducts.map(p => `
        <div class="product-card">
          <img src="${p.img}" class="product-img" data-src="${p.img}" alt="${brand}" loading="lazy">
          <div class="product-info">
            <div class="product-name">${p.name}</div>
            <div class="product-price">Private Quote</div>
          </div>
        </div>
      `).join('');
    }
  });
}

function getBrandId(name){
  const map = {"Cartier":"cartier","Vacheron Constantin":"vacheron","Rolex":"rolex","Patek Philippe":"patek","Longines":"longines","Omega":"omega","IWC":"iwc","Jaeger-LeCoultre":"jaeger","Audemars Piguet":"ap","Richard Mille":"richard","LV":"lv","Hermès":"hermes","Chanel":"chanel"};
  return map[name] || name.toLowerCase().replace(/\s+/g,'');
}

  setupSmoothNav();
}

  </div>`;
}

function goToBrand(brand) {
  currentBrand = brand;
  homePage.style.display = 'none';
  brandPage.style.display = 'block';
  brandTitle.innerText = brand;
  brandDesc.innerText = brandIntro[brand] || '';
  brandProductGrid.innerHTML = products.filter(p=>p.brand===brand).map(p=>createProductCard(p)).join('');

  brandPage.classList.remove('page-entrance');
  void brandPage.offsetWidth;

}

  homePage.classList.remove('page-entrance');

  homePage.classList.add('page-entrance');
}

function bindImageClick() {
  document.querySelectorAll('.product-img').forEach(img=>{
    img.onclick=(e)=>{
      e.stopPropagation();
      const rect = e.target.getBoundingClientRect();
      const originX = ((rect.left + rect.width/2) / window.innerWidth * 100).toFixed(2);
      const originY = ((rect.top + rect.height/2) / window.innerHeight * 100).toFixed(2);
      const modalBox = document.querySelector('#imageModal .modal-box');
      modalBox.style.transformOrigin = `${originX}% ${originY}%`;
      bigImage.src=img.dataset.src;
      imageModal.classList.add('show');
    };
  });
}

function closeImage() {
  const modal = document.getElementById('imageModal');
  modal.classList.remove('show'); // 关闭动画会自动触发缩小
}
